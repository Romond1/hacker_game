**Tech Stack:** React 19, TypeScript, Vite, Vitest, Testing Library, Node development API, PHP 8.1+, MySQL 8/MariaDB, existing CSS/effects/progression services.

```ts
export type TrainingRank = 'C' | 'B' | 'A' | 'S';
export type TrainingScoreRules = { basePerSuccess: number; errorPenalty: number; targetSeconds: number; timeBonus: number };
export type TrainingAggregate = { successes: number; errors: number; longestStreak: number; durationSeconds: number; totalRounds: number };
export type TrainingResult = { score: number; accuracy: number; longestStreak: number; rank: TrainingRank };
export type TrainingBest = { bestScore: number | null; bestTimeSeconds: number | null; bestAccuracy: number | null; longestStreak: number; highestRank: TrainingRank | null };
export type ValidationResult<E> = { valid: boolean; evidence: E; mistakes: number };
export type TrainingModuleDefinition<T, A, E> = {
  id: string; title: string; description: string; skill: string; linkedMissionId: string;
  rounds: number; difficulty: 'beginner'; generatorVersion: number; scoreRules: TrainingScoreRules;
  reward: { xpMax: number; creditsPerRun: number; creditCap: number };
  speedAchievementSeconds: number;
  generateTask(seed: number, roundIndex: number): T;
  validateTask(task: T, answer: A): ValidationResult<E>;
};

const rankOrder: TrainingRank[] = ['C', 'B', 'A', 'S'];
export function createSeededRandom(seed: number) {
  let state = seed >>> 0;
  return () => ((state = (Math.imul(1664525, state) + 1013904223) >>> 0) / 0x1_0000_0000);
}
export function calculateTrainingResult(rules: TrainingScoreRules, value: TrainingAggregate): TrainingResult {
  const attempts = value.successes + value.errors;
  const accuracy = attempts === 0 ? 0 : Math.round(value.successes / attempts * 100);
  const speed = value.durationSeconds <= rules.targetSeconds ? rules.timeBonus : Math.round(rules.timeBonus * rules.targetSeconds / value.durationSeconds);
  const score = Math.max(0, value.successes * rules.basePerSuccess - value.errors * rules.errorPenalty + speed);
  const rank: TrainingRank = accuracy === 100 && score >= 5000 ? 'S' : accuracy >= 90 ? 'A' : accuracy >= 75 ? 'B' : 'C';
  return { score, accuracy, longestStreak: value.longestStreak, rank };
}
export function improvesTrainingBest(best: TrainingBest, result: TrainingResult & { durationSeconds: number }): TrainingBest {
  return {
    bestScore: Math.max(best.bestScore ?? 0, result.score),
    bestTimeSeconds: best.bestTimeSeconds === null ? result.durationSeconds : Math.min(best.bestTimeSeconds, result.durationSeconds),
    bestAccuracy: Math.max(best.bestAccuracy ?? 0, result.accuracy),
    longestStreak: Math.max(best.longestStreak, result.longestStreak),
    highestRank: best.highestRank === null || rankOrder.indexOf(result.rank) > rankOrder.indexOf(best.highestRank) ? result.rank : best.highestRank,
  };
}
```

```ts
export type CalibrationTask = { prompt: string; choices: string[]; correctCode: string };
export type CalibrationEvidence = { selectedCode: string };
export const systemsCalibration: TrainingModuleDefinition<CalibrationTask, string, CalibrationEvidence> = {
  id: 'systems-calibration', title: 'Systems Calibration',
  description: 'Match each access code to verify the training systems.', skill: 'Visual matching',
  linkedMissionId: 'mission-3', rounds: 5, difficulty: 'beginner', generatorVersion: 1,
  scoreRules: { basePerSuccess: 900, errorPenalty: 200, targetSeconds: 30, timeBonus: 500 },
  reward: { xpMax: 150, creditsPerRun: 1, creditCap: 20 }, speedAchievementSeconds: 30,
  generateTask(seed, roundIndex) {
    const random = createSeededRandom(seed + roundIndex * 7919);
    const code = () => String(Math.floor(random() * 9000) + 1000);
    const correctCode = code();
    const choices = [correctCode, code(), code()].sort(() => random() - 0.5);
    return { prompt: `VERIFY ${correctCode}`, choices, correctCode };
  },
  validateTask(task, selectedCode) {
    return { valid: selectedCode === task.correctCode, evidence: { selectedCode }, mistakes: selectedCode === task.correctCode ? 0 : 1 };
  },
};
```
